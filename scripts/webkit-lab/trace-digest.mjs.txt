// Temporary (WebKit lab): what a failed journey's trace says about its requests, printed to the
// job's log, since nothing of a run can be downloaded from where the lab is read. Paths only —
// never a query or a fragment, where a link's token would be.
import { execFileSync } from 'node:child_process';
import { readdirSync, statSync } from 'node:fs';
import { join } from 'node:path';

const LAB_FAILURE = new RegExp(process.env.LAB_FAILURE ?? 'internal error', 'i');
const root = process.argv[2] ?? 'test-results';

function zips(dir) {
  const found = [];
  for (const name of readdirSync(dir)) {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) found.push(...zips(path));
    else if (name === 'trace.zip') found.push(path);
  }
  return found;
}

const unzip = (zip, args) =>
  execFileSync('unzip', [...args, zip], { maxBuffer: 1 << 30, encoding: 'utf8' });

const path = (address) => {
  try {
    const url = new URL(address);
    return `${url.port}${url.pathname}`;
  } catch {
    return '?';
  }
};
const tidy = (text) =>
  String(text)
    .replace(/\b(https?|wss?):\/\/([^\s'"/?#]*)([^\s'"?#]*)[^\s'"]*/g, '$1://$2$3')

    .replace(/[A-Za-z0-9_-]{24,}/g, '…')
    .replace(/\s+/g, ' ')
    .slice(0, 220);

function digest(zip) {
  const names = unzip(zip, ['-Z1']).split('\n').filter(Boolean);
  const events = [];
  for (const name of names.filter((entry) => /\.(trace|network)$/.test(entry))) {
    const context = name.replace(/-?trace\.(trace|network)$/, '') || 'main';
    for (const line of unzipOne(zip, name)) {
      let event;
      try {
        event = JSON.parse(line);
      } catch {
        continue;
      }
      if (event.type === 'resource-snapshot') {
        const snapshot = event.snapshot;
        const started = snapshot._monotonicTime;
        if (typeof started !== 'number') continue;
        const failure = snapshot.response?._failureText;
        events.push({
          at: started,
          end: started + Math.max(0, snapshot.time ?? 0),
          kind: 'request',
          context,
          text: `${snapshot.request?.method ?? '?'} ${path(snapshot.request?.url ?? '')}`,
          status: snapshot.response?.status ?? 0,
          failure: failure ?? null,
        });
      } else if (event.type === 'console') {
        if (event.messageType !== 'error' && event.messageType !== 'warning') continue;
        events.push({ at: event.time, kind: 'console', context, text: tidy(event.text) });
      } else if (event.type === 'before' && event.class !== 'Test') {
        const what = `${event.class ?? ''}.${event.method ?? ''}`;
        const on = event.params?.selector ?? event.params?.url ?? '';
        events.push({ at: event.startTime, kind: 'action', context, text: `${what} ${tidy(on)}` });
      } else if (event.type === 'event' && /crash|close/i.test(event.method ?? '')) {
        events.push({
          at: event.time,
          kind: 'event',
          context,
          text: `${event.class}.${event.method}`,
        });
      }
    }
  }
  events.sort((a, b) => a.at - b.at);
  return events;
}

function unzipOne(zip, name) {
  return execFileSync('unzip', ['-p', zip, name], { maxBuffer: 1 << 30, encoding: 'utf8' }).split(
    '\n',
  );
}

const seconds = (ms, zero) => ((ms - zero) / 1000).toFixed(2).padStart(7);

for (const zip of zips(root)) {
  console.log(`\n[trace] ${zip}`);
  let events;
  try {
    events = digest(zip);
  } catch (error) {
    console.log(`[trace]   unreadable: ${String(error).slice(0, 200)}`);
    continue;
  }
  if (events.length === 0) continue;
  const zero = events[0].at;
  const requests = events.filter((event) => event.kind === 'request');
  const failed = requests.filter((event) => event.failure !== null);
  const byFailure = new Map();
  for (const event of failed) byFailure.set(event.failure, (byFailure.get(event.failure) ?? 0) + 1);
  console.log(
    `[trace]   ${String(requests.length)} requests, ${String(failed.length)} failed: ${[...byFailure].map(([text, count]) => `${String(count)} × ${text}`).join('; ')}`,
  );
  // A crash of the network process fails what was in flight at one moment: such requests end
  // together. Each cluster of "internal error" endings is printed with what led up to it.
  const internal = failed
    .filter((event) => LAB_FAILURE.test(event.failure))
    .sort((a, b) => a.end - b.end);
  const clusters = [];
  for (const event of internal) {
    const last = clusters.at(-1);
    if (last && event.end - last.end < 1500) {
      last.end = event.end;
      last.count += 1;
    } else clusters.push({ start: event.end, end: event.end, count: 1 });
  }
  console.log(
    `[trace]   internal-error clusters: ${clusters.map((cluster) => `${String(cluster.count)} at ${seconds(cluster.start, zero).trim()}s`).join(', ') || 'none'}`,
  );
  for (const cluster of clusters.slice(0, 3)) {
    console.log(`[trace]   --- the cluster at ${seconds(cluster.start, zero).trim()}s`);
    const lost = internal.filter((event) => event.end >= cluster.start && event.end <= cluster.end);
    for (const event of lost.slice(0, 10)) {
      console.log(
        `[trace]   lost ${event.context.padEnd(4)} ${event.text} (sent ${seconds(event.at, zero).trim()}s)`,
      );
    }
    const before = requests.filter(
      (event) => event.at >= cluster.start - 10_000 && event.at <= cluster.start,
    );
    const perSecond = Array.from({ length: 10 }, (_, second) =>
      String(
        before.filter((event) => Math.floor((cluster.start - event.at) / 1000) === 9 - second)
          .length,
      ),
    );
    console.log(`[trace]   requests sent in each of the 10 s before: ${perSecond.join(' ')}`);
    const lastSecond = before.filter((event) => event.at >= cluster.start - 1000);
    console.log(
      `[trace]   in the last second: ${lastSecond
        .slice(-25)
        .map(
          (event) =>
            `${event.text.replace(/^(\w+) \d+\/api\/campaigns\/[^/]+/, '$1 …')}=${event.failure ?? String(event.status)}`,
        )
        .join(' | ')}`,
    );
    for (const event of events) {
      if (event.kind === 'request') continue;
      if (event.at < cluster.start - 10_000 || event.at > cluster.end + 2000) continue;
      console.log(
        `[trace]   ${seconds(event.at, zero)}s ${event.context.padEnd(4)} ${event.kind.padEnd(7)} ${event.text}`,
      );
    }
  }
}
